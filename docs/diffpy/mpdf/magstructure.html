  33class MagSpecies:
  34    """Store information for a single species of magnetic atom.
  35
  36
  37    This class takes a diffpy.Structure object and uses it to generate spins
  38    based on a set of propagation vectors and basis vectors. For more info
  39    about magnetic propagation vectors, visit e.g.
  40    http://dx.doi.org/10.1051/jp4:2001906 or Appendix B of Andrew Steele's
  41    dissertation at https://andrewsteele.co.uk/physics/thesis .
  42
  43    Args:
  44        struc (diffpy.Structure object): provides lattice parameters and unit
  45            cell of desired structure.
  46        label (string): label for this particular magnetic species. Should be
  47            different from the labels for any other magnetic species you make.
  48            Default is a random 8-character hexadecimal string.
  49        strucIdxs (python list): list of integers giving indices of magnetic
  50            atoms in the unit cell
  51        atoms (numpy array): list of atomic coordinates of all the magnetic
  52            atoms in the structure; e.g. generated by generateAtomsXYZ()
  53        spins (numpy array): triplets giving the spin vectors of all the
  54            atoms, in the same order as the atoms array provided as input.
  55            In units of hbar.
  56        calcIdxs (python list): list giving the indices of the atoms array
  57            specifying the atoms to be used as the origin when calculating
  58            the mPDF. If given the string argument 'all', then every atom
  59            will be used (potentially causing very long calculation times).
  60            These indices are relative to the atoms array for this specific
  61            MagSpecies, not relative to the atoms array for the
  62            MagStructure as a whole.
  63        rmaxAtoms (float): maximum distance from the origin of atomic
  64            positions generated by the makeAtoms method.
  65        avgmom (numpy array): three-vector giving the average magnetic moment
  66            for this species of magnetic atom to generate the spins. Relevant
  67            to incommensurate structures generated from an mcif.
  68        basisvecs (numpy array): nested three-vector(s) giving the basis
  69            vectors to generate the spins. e.g. np.array([[0, 0, 1]]). Any
  70            phase factor should be included directly with the basisvecs.
  71        kvecs (numpy array): nested three-vector(s) giving the propagation
  72            vectors for the magnetic structure in r.l.u.,
  73            e.g. np.array([[0.5, 0.5, 0.5]])
  74        S (float): Spin angular momentum quantum number in units of hbar.
  75        L (float): Orbital angular momentum quantum number in units of hbar.
  76        J (float): Total angular momentum quantum number in units of hbar.
  77        gS (float): spin component of the Lande g-factor (g = gS+gL).
  78            Calculated automatically from S, L, and J if not specified.
  79        gL (float): orbital component of the Lande g-factor. Calculated
  80            automatically from S, L, and J if not specified.
  81        g (float): Lande g-factor. Calculated automatically as gS+gL if not
  82            specified.
  83        j2type (string): Specifies the way the j2 integral is included in the
  84            magnetic form factor. Must be either 'RE' for rare earth or 'TM' for
  85            transition metal. If 'RE', the coefficient on the j2 integral is
  86            gL/g; if 'TM', the coefficient is (g-2)/g. Default is 'RE'. Note
  87            that for the default values of S, L, and J, we will have gS=2,
  88            gL=0, and g=2, and there will be no difference in the form factor
  89            calculation for 'RE' or 'TM'.
  90        ffparamkey (string): gives the appropriate key for getFFparams()
  91        ffqgrid (numpy array): grid of momentum transfer values used for
  92            calculating the magnetic form factor.
  93        ff (numpy array): magnetic form factor.
  94        useDiffpyStruc (boolean): True if atoms/spins to be generated from
  95            a diffpy structure object; False if a user-provided unit cell is
  96            to be used. Note that there may be some problems with user-
  97            provided unit cells with lattice angles strongly deviated from
  98            90 degrees.
  99        latVecs (numpy array): Provides the unit cell lattice vectors as
 100            np.array([avec, bvec, cvec]). Only useful if useDiffpyStruc = False.
 101        atomBasis (numpy array): Provides positions of the magnetic atoms
 102            in fractional coordinates within the unit cell. Only useful if
 103            useDiffpyStruc = False. Example: np.array([[0, 0, 0], [0.5, 0.5, 0.5]])
 104        spinBasis (numpy array): Provides the orientations of the spins in
 105            the unit cell, in the same order as atomBasis. Only useful if
 106            useDiffpyStruc = False. Example: np.array([[0, 0, 1], [0, 0, -1]]
 107        origin (numpy array): Cartesian coordinates of the position that will
 108            be considered the origin when generating spin directions from basis
 109            vectors and propagation vectors. Default is np.array([0,0,0]).
 110        verbose (boolean): If True, will print messages relating to the structure.
 111            Useful for troubleshooting. Default is False.
 112        useOcc (boolean): If True, atomic site occupancies will be used to scale
 113            the magnetic moment vector. Default is False.
 114        occ (scalar): Occupancy of the magnetic atom associated with

 254    def makeFF(self):
 255        """Generate the magnetic form factor.
 256        """
 257        # g = self.gS+self.gL # move this up to initialization of MagSpecies
 258        if getFFparams(self.ffparamkey) != ['none']:
 259            if self.j2type == 'RE':
 260                self.ff = (jCalc(self.ffqgrid, getFFparams(self.ffparamkey))+
 261                           self.gL/self.g * jCalc(self.ffqgrid, getFFparams(self.ffparamkey), j2=True))
 262            elif self.j2type == 'TM':
 263                self.ff = (jCalc(self.ffqgrid, getFFparams(self.ffparamkey))+
 264                           (self.g-2)/self.g * jCalc(self.ffqgrid, getFFparams(self.ffparamkey), j2=True))
 265            else:
 266                print("j2type argument must be either 'RE' or 'TM'. Using generic magnetic form factor.")
 267                self.ff = jCalc(self.ffqgrid)
 268
 269        else:
 270            print('Using generic magnetic form factor.')
 271            self.ff = jCalc(self.ffqgrid)

 335    def runChecks(self):
 336        """Run some simple checks and raise a warning if a problem is found.
 337        """
 338        if self.verbose:        
 339            print(('Running checks for '+self.label+' MagSpecies object...\n'))
 340
 341        flagCount = 0
 342        flag = False
 343
 344        if self.useDiffpyStruc:
 345            # check that basisvecs and kvecs have same shape
 346            if self.kvecs.shape != self.basisvecs.shape:
 347                flag = True
 348            if flag:
 349                flagCount += 1
 350                if self.verbose:
 351                    print('kvecs and basisvecs must have the same dimensions.')
 352
 353        else:
 354            # check for improperlatVecs array
 355            if self.latVecs.shape != (3, 3):
 356                flag = True
 357            if flag:
 358                flagCount += 1
 359                if self.verbose:
 360                    print('latVecs array does not have the correct dimensions.')
 361                    print('It must be a 3 x 3 nested array.')
 362                    print('Example: np.array([[4, 0, 0], [0, 4, 0], [0, 0, 4]])')
 363            flag = False
 364
 365            # check for mismatched number of atoms and spins in basis
 366            if self.atomBasis.shape != self.spinBasis.shape:
 367                flag = True
 368            if flag:
 369                flagCount += 1
 370                if self.verbose:
 371                    print('atomBasis and spinBasis must have the same dimensions.')
 372
 373        # summarize results
 374        if flagCount == 0:
 375            if self.verbose:
 376                print('All MagSpecies() checks passed. No obvious problems found.\n')

 383class MagStructure:
 384    """Build on the diffpy.Structure class to include magnetic attributes.
 385
 386    This class takes a diffpy.Structure object and packages additional info
 387    relating to magnetic structure, which can then be fed to an MPDFcalculator
 388    object.
 389
 390    Args:
 391        struc (diffpy.Structure object): provides lattice parameters and unit
 392            cell of desired structure.
 393        atomic_struc (diffpy.Structure object): provides atomic structure inferred
 394            from the MCIF file. Only relevant if reading in MCIF file. If not initialized, it is
 395            assumed that either the magnetic and atomic unit cells are equivalent
 396            or spins were generated from the atomic unit cell using basis vectors.
 397        transform (string): string describing the transformation from the atomic
 398            cell basis vectors to the magnetic cell basis vector. Should be 
 399            present whenever atomic_struc is.
 400        species (python dictionary): dictionary of magnetic species in the
 401            structure. The values are MagSpecies objects.
 402        atoms (numpy array): list of atomic coordinates of all the magnetic
 403            atoms in the structure; e.g. generated by generateAtomsXYZ()
 404        spins (numpy array): triplets giving the spin vectors of all the
 405            atoms, in the same order as the atoms array provided as input.
 406        gfactors (numpy array): Lande g-factors of the magnetic moments
 407        rmaxAtoms (float): maximum distance from the origin of atomic
 408            positions generated by the makeAtoms method.
 409        ffqgrid (numpy array): grid of momentum transfer values used for
 410            calculating the magnetic form factor.
 411        ff (numpy array): magnetic form factor. Should be same shape as
 412            ffqgrid.
 413        label (string): Optional descriptive string for the MagStructure.
 414        K1 (float): a constant used for calculating Dr; should be averaged
 415            over all magnetic species. Important if physical information is
 416            to be extracted from mPDF scale factors, e.g. moment size.
 417        K2 (float): another constant used for calculating Dr.
 418        fractions (python dictionary): Dictionary providing the fraction of
 419            spins in the magnetic structure corresponding to each species.
 420        verbose (boolean): If True, will print messages relating to the structure.
 421            Useful for troubleshooting. Default is False.
 422        calcIdxs (python list): list giving the indices of the atoms array
 423            specifying the atoms to be used as the origin when calculating
 424            the mPDF. If given the string argument 'all', then every atom
 425            will be used (potentially causing very long calculation times).
 426        corrLength (scalar): magnetic correlation length such that the
 427            magnitude of the correlation between two spins separated by a
 428            distance d is given by exp(-d/corrLength). If set to zero, the
 429            correlation length is assumed to be infinite.
 430        dampingMat (3x3 matrix): damping matrix that encodes anisotropic
 431            correlation lengths. If nonzero, this supercedes the scalar
 432            corrLength attribute (which can only be used for isotropic
 433            correlation lengths).
 434        rho0 (float): number of magnetic moments per cubic Angstrom in the
 435            magnetic structure; default value is 0.
 436        netMag (float): net magnetization in Bohr magnetons per magnetic moment
 437            in the sample; default is 0. Only nonzero for ferro/ferrimagnets or

 516    def makeSpecies(self, label, strucIdxs=None, atoms=None, spins=None,
 517                    basisvecs=None, kvecs=None, S=0.5, L=0.0, J=None, gS=None,
 518                    gL=None, g=None, j2type=None, ffparamkey=None,ffqgrid=None,
 519                    ff=None, occ=None):
 520        """Create a MagSpecies object and add it to the species dictionary.
 521
 522        Args:
 523            label (string): label for this particular magnetic species. Should be
 524                different from the labels for any other magnetic species you make.
 525            strucIdxs (python list): list of integers giving indices of magnetic
 526                atoms in the unit cell
 527            atoms (numpy array): list of atomic coordinates of all the magnetic
 528                atoms in the structure; e.g. generated by generateAtomsXYZ()
 529            spins (numpy array): triplets giving the spin vectors of all the
 530                atoms, in the same order as the atoms array provided as input.
 531            basisvecs (numpy array): nested three-vector(s) giving the basis
 532                vectors to generate the spins. e.g. np.array([[0, 0, 1]]). Any
 533                phase factor should be included directly with the basisvecs.
 534            kvecs (numpy array): nested three-vector(s) giving the propagation
 535                vectors for the magnetic structure in r.l.u.,
 536                e.g. np.array([[0.5, 0.5, 0.5]])
 537            gS (float): spin component of the Lande g-factor (g = gS+gL).
 538                Calculated automatically from S, L, and J if not specified.
 539            gL (float): orbital component of the Lande g-factor. Calculated
 540                automatically from S, L, and J if not specified.
 541            g (float): Lande g-factor. Calculated automatically as gS+gL if not
 542                specified.
 543            j2type (string): Specifies the way the j2 integral is included in the
 544                magnetic form factor. Must be either 'RE' for rare earth or 'TM' for
 545                transition metal. If 'RE', the coefficient on the j2 integral is
 546                gL/g; if 'TM', the coefficient is (g-2)/g. Default is 'RE'. Note
 547                that for the default values of S, L, and J, we will have gS=2,
 548                gL=0, and g=2, and there will be no difference in the form factor
 549                calculation for 'RE' or 'TM'.
 550            ffparamkey (string): gives the appropriate key for getFFparams()
 551            ffqgrid (numpy array): grid of momentum transfer values used for
 552                calculating the magnetic form factor.
 553            ff (numpy array): magnetic form factor.
 554
 555        """

 961        """Apply a correlation length to the spin magnitudes.
 962
 963        Args:
 964            originIdx (int): The index of the spin in magstructure.spins
 965                that should be considered the origin.
 966
 967        Returns:
 968            scaledSpins (np.array): An array with the same shape as the
 969                self.spins array, where the magnitudes of the spins have
 970                been scaled in accordance with the correlation length.
 971        """
 972        scaledSpins = 1.0*self.spins
 973        if type(self.dampingMat) != np.ndarray: # isotropic correlation length
 974            xi = self.corrLength
 975            if xi != 0.0:
 976                distanceVecs = self.atoms - self.atoms[originIdx]
 977                distances = np.apply_along_axis(np.linalg.norm, 1, distanceVecs)
 978                rescale = np.exp(-distances/xi)[:,np.newaxis] 
 979                scaledSpins *= rescale 
 980        else:
 981            dampingMat = self.dampingMat
 982            # ensure that dampingMat is 3x3 symmetric matrix
 983            if dampingMat.shape == (3, 3):
 984                if np.allclose(dampingMat, dampingMat.T, rtol=1e-5, atol=1e-8):
 985                    distanceVecs = self.atoms - self.atoms[originIdx]
 986                    distances = np.apply_along_axis(np.linalg.norm, 1, distanceVecs)
 987                    distanceVecsN = distanceVecs/np.apply_along_axis(np.linalg.norm,1,distanceVecs)[:,np.newaxis]
 988                    mult1 = np.tensordot(dampingMat, distanceVecsN, axes=(0,1)).T
 989                    xi = 1.0/np.sqrt(np.apply_along_axis(np.sum, 1, distanceVecsN*mult1))
 990                    xi[originIdx] = 1.0 # avoid divide-by-zero problem
 991                    rescale = np.exp(-distances/xi)[:,np.newaxis] 
 992                    scaledSpins *= rescale                     
 993                else:
 994                    print('Damping matrix is not symmetric. Spins will not be modified.')
 995            else:
 996                print('Damping matrix is not a 3x3 matrix. Spins will not be modified.')
