 36def create_from_mcif(mcif, ffparamkey=None, rmaxAtoms=20, S=0.5, L=0.0,
 37                     J=None, gS=None, gL=None, g=None, j2type=None, occ=None):
 38    """
 39    Creates a MagStructure object from an MCIF file.
 40    
 41    Note: The only acceptable incommensurate structures at the moment are 1-k
 42    structures with no Fourier harmonics (i.e. only coefficients for
 43    n = 1) and no atoms with nonzero average magnetic moment.
 44
 45    Args:
 46        mcif (string): path to MCIF file for desired magnetic structure.
 47        ffparamkey (string): optional; gives the appropriate key for getFFparams()
 48            to generate the correct magnetic form factor.
 49        rmaxAtoms (float): radius to which magnetic atoms should be generated.
 50            Default is 20 Angstroms.
 51        S (float): Spin angular momentum quantum number in units of hbar.
 52        L (float): Orbital angular momentum quantum number in units of hbar.
 53        J (float): Total angular momentum quantum number in units of hbar.
 54        gS (float): spin component of the Lande g-factor (g = gS+gL).
 55            Calculated automatically from S, L, and J if not specified.
 56        gL (float): orbital component of the Lande g-factor. Calculated
 57            automatically from S, L, and J if not specified.
 58        g (float): Lande g-factor. Calculated automatically as gS+gL if not
 59            specified.
 60        j2type (string): Specifies the way the j2 integral is included in the
 61            magnetic form factor. Must be either 'RE' for rare earth or 'TM' for
 62            transition metal. If 'RE', the coefficient on the j2 integral is
 63            gL/g; if 'TM', the coefficient is (g-2)/g. Default is 'RE'. Note
 64            that for the default values of S, L, and J, we will have gS=2,
 65            gL=0, and g=2, and there will be no difference in the form factor
 66            calculation for 'RE' or 'TM'.
 67
 68    Returns:
 69        MagStructure object corresponding to the magnetic structure
 70            encoded in the MCIF file. Note that the position and spin arrays
 71            are not automatically populated when using this function, so
 72            MagStructure.makeAll() will likely need to be called afterward.
 73            Also note that variables such as ffparamkey, rmaxAtoms, S, L, J,
 74            etc are applied uniformly to all magnetic atoms in the unit cell,
 75            so this method cannot create MagSpecies objects for which these
 76            attributes are different.
 77
 78    """
 79    # creates an empty MagStructure
 80    mstruc = MagStructure()    
 81
 82    #Flags whether the structure is incommensurate
 83    incomm = False
 84
 85    #Extracts the needed information from the mcif file
 86    tags = {}
 87    parser = SimpleParser(tags)
 88    tags = parser.ReadFile(mcif)
 89
 90    #These are the parameters of the basic or unit magnetic cell
 91    #'a', 'b', and 'c' are in angstroms, while 'alpha', 'beta', and 'gamma' are in degrees
 92    a = tags['_cell_length_a'][0]
 93    b = tags['_cell_length_b'][0]
 94    c = tags['_cell_length_c'][0]
 95    alpha = tags['_cell_angle_alpha'][0]
 96    beta = tags['_cell_angle_beta'][0]
 97    gamma = tags['_cell_angle_gamma'][0]
 98    transform = tags['_parent_space_group.child_transform_Pp_abc'][0]
 99
100    #symbols = tags['_atom_site_type_symbol']
101    symbols = tags['_atom_site_label']
102    symbols0 = 1*symbols # starting list of atom symbols
103    try:
104        occs = tags['_atom_site_occupancy']
105    except KeyError:
106        occs = [1.0] * len(symbols)
107    occs0 = 1*occs # starting list of occupancies
108    scaled_pos = np.array([tags['_atom_site_fract_x'],
109                           tags['_atom_site_fract_y'],
110                           tags['_atom_site_fract_z']]).T
111    scaled_pos = np.mod(scaled_pos,1.)
112
113    mag_mom = np.zeros_like(scaled_pos)
114    mag_idx = []
115
116    if '_cell_wave_vector_x' in tags:
117        #This is the branch for incommensurate structures
118        incomm = True
119        k = np.matrix.flatten(np.array([tags['_cell_wave_vector_x'],
120                                        tags['_cell_wave_vector_y'],
121                                        tags['_cell_wave_vector_z']]).T)
122
123        symm_cent = tags['_space_group_symop_magn_ssg_centering.algebraic']
124        symm_ops = tags['_space_group_symop_magn_ssg_operation.algebraic']
125
126        #For incommensurate structures, mag_mom is the average magnetic moment of an atom. Currently,
127        #this variable is unused for incommensurate structures
128        four_cos = np.zeros_like(scaled_pos)
129        four_sin = np.zeros_like(scaled_pos)
130        for i, label in enumerate(tags['_atom_site_label']):
131            if label in tags['_atom_site_moment.label']:
132                mag_idx.append(i)
133                mi = tags['_atom_site_moment.label'].index(label)
134                mag_mom[i] = [tags['_atom_site_moment.crystalaxis_x'][mi],
135                              tags['_atom_site_moment.crystalaxis_y'][mi],
136                              tags['_atom_site_moment.crystalaxis_z'][mi]]
137
138                four_cos[i] = [tags['_atom_site_moment_Fourier_param.cos'][3*mi],
139                               tags['_atom_site_moment_Fourier_param.cos'][3*mi + 1],
140                               tags['_atom_site_moment_Fourier_param.cos'][3*mi + 2]]
141
142                four_sin[i] = [tags['_atom_site_moment_Fourier_param.sin'][3*mi],
143                               tags['_atom_site_moment_Fourier_param.sin'][3*mi + 1],
144                               tags['_atom_site_moment_Fourier_param.sin'][3*mi + 2]]
145    else:
146        #This is the branch for commensurate structures
147        k = np.array([0,0,0])
148
149        symm_cent = tags['_space_group_symop_magn_centering.xyz']
150        symm_ops = tags['_space_group_symop_magn_operation.xyz']
151
152        for i, label in enumerate(tags['_atom_site_label']):
153            if label in tags['_atom_site_moment.label']:
154                mag_idx.append(i)
155                mi = tags['_atom_site_moment.label'].index(label)
156                mag_mom[i] = [tags['_atom_site_moment.crystalaxis_x'][mi],
157                              tags['_atom_site_moment.crystalaxis_y'][mi],
158                              tags['_atom_site_moment.crystalaxis_z'][mi]]
159
160    #Takes the magnetic information to reduced lattice coordinates
161    LL = np.diag([1./a,1./b,1./c])
162    mag_mom = dot(mag_mom,LL)
163    if incomm:
164        four_cos = dot(four_cos,LL)
165        four_sin = dot(four_sin,LL)
166
167    #Performs the symmetry operations to populate the rest of the unit cell
168    all_pos = np.copy(scaled_pos)
169    all_mom = np.copy(mag_mom)
170    if incomm:
171        all_cos = np.copy(four_cos)
172        all_sin = np.copy(four_sin)
173
174    tol = 1e-3
175    for j, pos in enumerate(scaled_pos):
176        pos_subset = np.array([pos])
177        eq_site_counter = 1
178        for cent in symm_cent:
179            #Applies a centering operation
180            if incomm:
181                Rc, hc, r1c, tc, t4c, trc = parse_magn_ssg_operation(cent)
182            else:
183                Rc, tc, trc = parse_magn_operation(cent)
184            cent_pos = np.mod(dot(Rc,pos) + tc,1.)
185
186            for op in symm_ops:
187                #Applies a symmetry operation
188                if incomm:
189                    R, h, r1, t, t4, tr = parse_magn_ssg_operation(op)
190                else:
191                    R, t, tr = parse_magn_operation(op)
192                eq_site = dot(R,cent_pos) + t
193
194                #Handles the cases where eq_site has coordinate values extremely close to 0 or 1
195                for l, coord in enumerate(eq_site):
196                    if abs(1 - coord) < tol:
197                        eq_site[l] = 1.
198                    if abs(coord) < tol:
199                        eq_site[l] = 0.
200                eq_site = np.mod(eq_site,1.)
201                eq_site = np.round(eq_site, decimals=6)
202
203                #Checks to make sure site hasn't already been occupied
204                for l, pp in enumerate(pos_subset):
205                    if np.allclose(eq_site,pp,atol=tol): # and the atom labels are the same (fix this)
206                        break
207                else:
208                    #Adds the new site found with symmetry
209                    if j in mag_idx:
210                        mag_idx.append(len(symbols))
211                    symbols.append(symbols0[j]+'.'+str(eq_site_counter))
212                    occs.append(occs0[j])
213                    eq_site_counter += 1
214                    pos_subset = np.append(pos_subset,[eq_site],axis=0)
215                    all_pos = np.append(all_pos,[eq_site],axis=0)
216
217                    #Transforms the moment information
218                    if incomm:
219                        deltc = 2*pi*(t4c + dot(hc,pos))
220                        delt = 2*pi*(t4 + dot(h,cent_pos))
221
222                        cent_cos = trc*det(Rc)*(cos(deltc)*dot(Rc,four_cos[j]) - r1c*sin(deltc)*dot(Rc,four_sin[j]))
223                        cent_sin = trc*det(Rc)*(sin(deltc)*dot(Rc,four_cos[j]) + r1c*cos(deltc)*dot(Rc,four_sin[j]))
224
225                        new_cos = tr*det(R)*(cos(delt)*dot(R,cent_cos) - r1*sin(delt)*dot(R,cent_sin))
226                        new_sin = tr*det(R)*(sin(delt)*dot(R,cent_cos) + r1*cos(delt)*dot(R,cent_sin))
227
228                        all_cos = np.append(all_cos,[new_cos],axis=0)
229                        all_sin = np.append(all_sin,[new_sin],axis=0)
230
231                    new_mom = tr*trc*det(R)*det(Rc)*dot(R,dot(Rc,mag_mom[j]))   
232                    all_mom = np.append(all_mom,[new_mom],axis=0)
233
234    #Determines the basis vectors of the structure (same as the magnetic moments if commensurate)
235    basis_vecs = np.empty(np.shape(all_pos),dtype=complex)
236    if incomm:
237        basis_vecs = (0.5*np.exp(-2*pi*1j*dot(all_pos,k)))[:,np.newaxis]*(all_cos + 1j*all_sin)
238        #for i, pos in enumerate(all_pos):
239            #basis_vecs[i] = 0.5*exp(-2*pi*1j*dot(k,pos))*(all_cos[i] + 1j*all_sin[i])
240    else:
241        basis_vecs = all_mom.astype(complex)
242        #for i, pos in enumerate(all_pos):
243            #basis_vecs[i] = all_mom[i].astype(complex)
244
245    #Converts the basis vectors to Cartesian coordinates
246    lat = Lattice(a,b,c,alpha,beta,gamma)
247    basis_vecs = dot(basis_vecs,lat.base)
248    all_mom = dot(all_mom,lat.base)
249
250    #Creates a diffpy Structure object
251    atoms = []
252    for i, pos in enumerate(all_pos):
253        atoms.append(Atom(atype=symbols[i],xyz=pos, occupancy=occs[i]))
254    astruc = Structure(atoms=atoms)
255    astruc.lattice = lat
256    for atom in astruc: # remove extra numbers from atom names
257        name = atom.element
258        newname = re.findall(r'[a-zA-Z]+', name)[0]
259        atom.element = newname
260
261
262    # Determine Lande g factor to convert the moment size (stored in mcif) to momentum size
263    if J is None:
264        J = S + L
265    if gS is None:
266        gS = 1.0 + 1.0*(S*(S+1)-L*(L+1))/(J*(J+1))
267    if gL is None:
268        gL = 0.5 + 1.0*(L*(L+1)-S*(S+1))/(2*J*(J+1))
269    if g is None:
270        g = gS + gL
271    
272    # Creates a separate MagSpecies object for each magnetic atom in the unit cell
273    for idx in mag_idx:
274        new_mspec = MagSpecies(ffparamkey=ffparamkey, rmaxAtoms=rmaxAtoms,
275                               S=S, L=L, J=J, gS=gS, gL=gL, g=g,
276                               j2type=j2type)
277
278        new_mspec.struc = astruc
279        new_mspec.label = symbols[idx]
280
281        new_mspec.strucIdxs = [idx]
282        new_mspec.origin = all_pos[idx]
283        new_mspec.occ = occs[idx]
284
285        new_mspec.kvecs = np.array([k])
286        new_mspec.basisvecs = np.array([basis_vecs[idx]])
287        if incomm:
288            new_mspec.kvecs = np.append(new_mspec.kvecs,[-k],axis=0)
289            new_mspec.basisvecs = np.append(new_mspec.basisvecs,[np.conjugate(basis_vecs[idx])],axis=0)
290        new_mspec.basisvecs /= g # convert to angular momentum rather than magnetic moment
291        new_mspec.avgmom = np.array(all_mom[idx]) if incomm else np.array([0, 0, 0])
292        new_mspec.avgmom = np.array(new_mspec.avgmom) / g
293
294        mstruc.loadSpecies(new_mspec)
295
296    mstruc.transform = transform
297    print('MagStructure creation from mcif file successful.')
298    return mstruc

302def create_atomic_cell(mstruc,transform):
303    """
304    This method accepts a MagStructure object that has already been loaded with an mcif file
305    and a string containing the parent-child basis transformation information. The output will be a new
306    diffpy.Structure object loaded with the atomic structure information related to the magnetic structure
307    of the original sample.
308    """
309    #Processes the string with the parent-child transformation info
310    R, t = parse_parent_child_transform(transform)
311    R_p = inv(R)
312    t_p = dot(t,R_p)
313
314    #Transforms the magnetic cell basis to an atomic cell basis per the inverse of the given transform
315    mag_cell = mstruc.struc.lattice.base
316    atom_cell = dot(R_p,mag_cell)
317
318    #Determines the dimension of the supercell we need to completely contain the atomic cell
319    dim  = [[floor(min(R_p[:,0]) - t_p[0]) - 1,ceil(max(R_p[:,0]) - t_p[0]) + 1],
320            [floor(min(R_p[:,1]) - t_p[1]) - 1,ceil(max(R_p[:,1]) - t_p[1]) + 1],
321            [floor(min(R_p[:,2]) - t_p[2]) - 1,ceil(max(R_p[:,2]) - t_p[2]) + 1]]
322
323    sup_pos, sup_occup, sup_symb = supersize_me(mstruc.struc,dim)
324
325    #Determines atomic unit cell from the newly created magnetic supercell
326    scaled_pos = []
327    occup = []
328    symb = []
329
330    tol = 1e-3
331    for l, pos in enumerate(sup_pos):
332        #Converts the position to fractional atomic lattice coordinates
333        new_pos = dot(pos,R) + t
334
335        #Tidies up coordinates close enough to 0 or 1        
336        for i, coord in enumerate(new_pos):
337            if abs(1 - coord) < tol:
338                new_pos[i] = 1.
339            if abs(coord) < tol:
340                new_pos[i] = 0.
341
342        #Checks if the atoms sits inside the atomic unit cell
343        if (0<=new_pos[0]<1) and (0<=new_pos[1]<1) and (0<=new_pos[2]<1):
344            add_pos = True            
345
346            #Also checks for duplicates
347            dupes, dupe_idxs = find_dupes(scaled_pos,new_pos,tol)
348            if len(dupes) >= 1:
349                tot_occup = sup_occup[l]
350                for idx in dupe_idxs:
351                    tot_occup += occup[idx]
352                    if element(symb[idx]) == element(sup_symb[l]):
353                        add_pos = False
354                if (tot_occup - 1) > tol:
355                    add_pos = False
356            #for m, pp in enumerate(scaled_pos):
357            #    if np.allclose(new_pos,pp,atol=tol):
358            #        break
359            #else:
360            if add_pos:
361                scaled_pos.append(new_pos)
362                occup.append(sup_occup[l])
363                symb.append(sup_symb[l])
364
365    #Creates a new diffpy.Structure object representing the atomic unit cell
366    atoms = []
367    for l, pos in enumerate(scaled_pos):
368        atoms.append(Atom(atype=symb[l],xyz=pos,occupancy=occup[l]))
369    new_struc = Structure(atoms=atoms)
370    new_struc.lattice = Lattice(base=atom_cell)
371
372    return new_struc

375def create_magnetic_cell(astruc,transform):
376    """
377    Takes a diffpy.Structure object representing the atomic unit cell and a string containing information 
378    about the parent to child basis vector transformation. Returns a new diffpy.Structure object
379    representing the magnetic unit cell. 
380
381    Note: At the moment, the resulting magnetic structure won't actually have any magnetic moment 
382    information; it will just be a list of atoms and their positions in the unit cell. Work still needs to
383    be done to figure out how to connect atoms in the magnetic cell with the correct moment information.
384    """
385    #Processes the string with the parent-child transformation info
386    R, t = parse_parent_child_transform(transform)
387    R_p = inv(R)
388    t_p = dot(t,R_p)
389    
390    #Transforms the atomic cell basis to a magnetic cell basis using the given transform
391    atom_cell = astruc.lattice.base
392    mag_cell = dot(R,atom_cell)
393    
394    #Determines the dimensions of the supercell we want to work with to get the magnetic cell
395    dim = [[floor(min(R[:,0]) + t[0]) - 1,ceil(max(R[:,0]) + t[0]) + 1],
396           [floor(min(R[:,1]) + t[1]) - 1,ceil(max(R[:,1]) + t[1]) + 1],
397           [floor(min(R[:,2]) + t[2]) - 1,ceil(max(R[:,2]) + t[2]) + 1]]
398    
399    sup_pos, sup_occup, sup_symb = supersize_me(astruc,dim)
400    
401    #Picks out the magnetic unit cell from the newly created atomic supercell
402    scaled_pos = []
403    occup = []
404    symb = []
405    
406    tol = 1e-3
407    for l, pos in enumerate(sup_pos):
408        #Converts the position to fractional magnetic lattice coordinates
409        new_pos = dot(pos,R_p) - t_p
410
411        #Tidies up coordinates close enough to 0 or 1        
412        for i, coord in enumerate(new_pos):
413            if abs(1 - coord) < tol:
414                new_pos[i] = 1.
415            if abs(coord) < tol:
416                new_pos[i] = 0.
417
418        #Checks if the atom sits inside the magnetic unit cell
419        if (0<=new_pos[0]<1) and (0<=new_pos[1]<1) and (0<=new_pos[2]<1):
420            add_pos = True
421
422            #Also checks for duplicates (which may be allowed based on occupancy)
423            dupes, dupe_idxs = find_dupes(scaled_pos,new_pos,tol)
424            if len(dupes) >= 1:
425                tot_occup = sup_occup[l]
426                for idx in dupe_idxs:
427                    tot_occup += occup[idx]
428                    if element(symb[idx]) == element(sup_symb[l]):
429                        add_pos = False
430                if (tot_occup - 1) > tol:
431                    add_pos = False
432            #for m, pp in enumerate(scaled_pos):
433            #    if np.allclose(new_pos,pp,atol=tol):
434            #        break
435            #else:
436            if add_pos:
437                scaled_pos.append(new_pos)
438                occup.append(sup_occup[l])
439                symb.append(sup_symb[l])
440            
441    #Creates a new diffpy.Structure object representing the magnetic unit cell (without the moment information)
442    atoms = []
443    for l, pos in enumerate(scaled_pos):
444        atoms.append(Atom(atype=symb[l],xyz=pos,occupancy=occup[l]))
445    new_struc = Structure(atoms=atoms)
446    new_struc.lattice = Lattice(base=mag_cell)
447    
448    return new_struc

451def parse_magn_ssg_operation(op):
452    """
453    Parses a string containing a superspace group symmetry operation in standard
454    notation. Returns the rotation and translation part of the associated space
455    group operation (R and t), the average structure reciprocal lattice vector (h), 
456    superspace phase information (r1 and t4), and the time reversal sign (p).
457    """
458    r1,r2,r3,r4,p=op.split(',')
459    M = np.zeros([4,4])
460    t = np.zeros([3])
461    h = np.zeros([3])
462    
463    #Compiles translation information
464    t[0] = convert_to_float(r1[-4:]) if '/' in r1 else 0.
465    t[1] = convert_to_float(r2[-4:]) if '/' in r2 else 0.
466    t[2] = convert_to_float(r3[-4:]) if '/' in r3 else 0.
467    t4 = convert_to_float(r4[-4:]) if '/' in r4 else 0.
468    
469    #Compiles rotation matrix
470    for i, line in enumerate([r1,r2,r3,r4]):
471        x1=x2=x3=x4=0.
472        m = re.search(r'-?\d?x1',line)
473        if m:
474            xs=m.group()
475            x1 = 1. if ('x1' in xs and not '-' in xs) else -1.
476            xs = xs.replace('x1','')
477            m = re.search(r'\d',xs)
478            if m:
479                x1 *= float(m.group())
480
481        m = re.search(r'-?\d?x2',line)
482        if m:
483            ys=m.group()
484            x2 = 1. if ('x2' in ys and not '-' in ys) else -1.
485            ys = ys.replace('x2','')
486            m = re.search(r'\d',ys)
487            if m:
488                x2 *= float(m.group())
489                
490        m = re.search(r'-?\d?x3',line)
491        if m:
492            zs=m.group()
493            x3 = 1. if ('x3' in zs and not '-' in zs) else -1.
494            zs = zs.replace('x3','')
495            m = re.search(r'\d',zs)
496            if m:
497                x3 *= float(m.group())
498                
499        m = re.search(r'-?\d?x4',line)
500        if m:
501            ts = m.group()
502            x4 = 1. if ('x4' in ts and not '-' in ts) else -1.
503            ts = ts.replace('x4','')
504            m = re.search(r'\d',ts)
505            if m:
506                x4 *= float(m.group())
507            
508        M[i,0], M[i,1], M[i,2], M[i,3] = [x1,x2,x3,x4]
509    
510    R = M[:3,:3]
511    h = M[3,:3]
512    r1 = M[3,3]
513        
514    return (R,h,r1,t,t4,float(p))

517#Essentially a copy of muesr's parse_magn_operation_xyz_string function
518def parse_magn_operation(op):
519    """
520    Parses a string containg a space group symmetry operation in standard notation.
521    Returns the rotation and translation parts of the operation (R and t) and the 
522    time reversal sign (p).
523    """
524    r1,r2,r3,p=op.split(',')
525    R = np.zeros([3,3]) # rotations
526    t = np.zeros([3])   # translations
527    
528    # compile traslation matrix

664def order_by_pos(original,unordered):
665    """
666    This method accepts two arrays of the same size and outputs a third array also of that size.
667    The third array is created by permuting the rows of the 'unordered' array in such a way
668    that the point given by the ith row of the output array is closer (in terms of Euclidean distance)
669    to the point given by the ith row vector of the 'original' array than to the point given by any
670    other row of the 'original' array. The intended application of this method is to recover the
671    ordering of an array of atomic positions in a magnetic unit cell after downsizing to an atomic unit
672    cell and then upsizing once more.
673    
674    In this method, we're assuming that each point in the 'unordered' array is closest to exactly one 
675    point in the original array. The motivation for this is that the rescaling of the atomic unit cell
676    during PDF and mPDF fits should only cause small changes in atomic positions within the unit cell, 
677    so no atom should stray far enough from their original position to end up closer to a different
678    atom.
679    """
680    #Checks if the input arrays have the same shape
681    if np.shape(original)!=np.shape(unordered):
682        print('Error: Input arrays must be of the same size.')
683        return
684
685    ordered = np.zeros_like(unordered)
686    unordered_copy = np.copy(unordered)
687
688    for i, pos_orig in enumerate(original):
689        #For a given point in the 'original' array, finds the closest point in the 'unordered' array
690        closest_pos = 0
691        closest_dist = 1e6
692
693        for j, pos_unord in enumerate(unordered_copy):
694            curr_dist = norm(pos_orig - pos_unord)
695            if curr_dist < closest_dist:
696                closest_dist = curr_dist
697                closest_pos = j
698
699        #Adds the closest point in the 'unordered' array to the 'ordered' array in the correct position
700        ordered[i] = unordered_copy[closest_pos]
701
702        #Deletes the position from the 'unordered' array to speed up computation
703        unordered_copy = np.delete(unordered_copy,closest_pos,0)
704
705    return ordered

765    cx = cos(beta)

771    cell = np.vstack((va,vb,vc))
772    return cell
