 24class MPDFcalculator:
 25    """Create an MPDFcalculator object to help calculate mPDF functions.
 26
 27    This class is loosely modelled after the PDFcalculator class in diffpy.
 28    At minimum, it requires a magnetic structure with atoms and spins, and
 29    it calculates the mPDF from that. Various other options can be specified
 30    for the calculated mPDF.
 31
 32    Args:
 33        magstruc (MagStructure object): provides information about the
 34            magnetic structure. Must have arrays of atoms and spins.
 35        extendedrmin (float): extension of the r-grid on which the mPDF is
 36            calculated to properly account for contribution of pairs just
 37            before the boundary. 4 A by default.
 38        extendedrmax (float): extension of the r-grid on which the mPDF is
 39            calculated to properly account for contribution of pairs just
 40            outside the boundary. 4 A by default.
 41        qdamp (float): usual PDF qdamp parameter. Turned off if set to zero.
 42        qmin (float): minimum experimentally accessible q-value (to be used
 43            for simulating termination ripples). If <0, no termination effects
 44            are included.
 45        qmax (float): maximum experimentally accessible q-value (to be used
 46            for simulating termination ripples). If <0, no termination effects
 47            are included.
 48        rmin (float): minimum value of r for which mPDF should be calculated.
 49        rmax (float): maximum value of r for which mPDF should be calculated.
 50        rstep (float): step size for r-grid of calculated mPDF.
 51        ordScale (float): overall scale factor for the mPDF function f(r).
 52        paraScale (float): scale factor for the paramagnetic part of the
 53            unnormalized mPDF function D(r).
 54        rmintr (float): minimum value of r for the Fourier transform of the
 55            magnetic form factor required for unnormalized mPDF.
 56        rmaxtr (float): maximum value of r for the Fourier transform of the
 57            magnetic form factor required for unnormalized mPDF.
 58        drtr (float): step size for r-grid used for calculating Fourier
 59            transform of magnetic form mactor.
 60        label (string): Optional descriptive string for the MPDFcalculator.
 61        qwindow (numpy array): Q-space window function applied to the data
 62            prior to Fourier transformation. Not used by default.
 63        qgrid (numpy array): Q-space grid on which the window function is
 64            defined.
 65        """
 66    def __init__(self, magstruc=None, extendedrmax=4.0,
 67                 extendedrmin=4.0, qdamp=0.0, qmin=0.0,
 68                 qmax=-1.0, rmin=0.0, rmax=20.0, rstep=0.01,
 69                 ordScale=1.0, paraScale=1.0, rmintr=-5.0,
 70                 rmaxtr=5.0, label='', qwindow=None, qgrid=None):
 71        if magstruc is None:
 72            self.magstruc = []
 73        else:
 74            self.magstruc = magstruc
 75            if magstruc.rmaxAtoms < rmax:
 76                print('Warning: Your structure may not be big enough for your')
 77                print('desired calculation range.')
 78        self.extendedrmin = extendedrmin
 79        self.extendedrmax = extendedrmax
 80        self.qdamp = qdamp
 81        self.qmin = qmin
 82        self.qmax = qmax
 83        self.rmin = rmin
 84        self.rmax = rmax
 85        self.rstep = rstep
 86        self.ordScale = ordScale
 87        self.paraScale = paraScale
 88        self.rmintr = rmintr
 89        self.rmaxtr = rmaxtr
 90        self.label = label
 91        if qwindow is None:
 92            self.qwindow = np.array([0])
 93        else:
 94            self.qwindow = qwindow
 95        if qgrid is None:
 96            self.qgrid = np.array([0])
 97        else:
 98            self.qgrid = qgrid

106    def calc(self, normalized=True, both=False, correlationMethod='simple',
107             linearTermMethod='exact'):
108        """Calculate the magnetic PDF.
109
110        Args:
111            normalized (boolean): indicates whether or not the normalized mPDF
112                should be returned.
113            both (boolean): indicates whether or not both normalized and
114                unnormalized mPDF quantities should be returned.
115            correlationMethod (string): determines how the calculation should
116                be done if the correlation length is finite. Options are:
117                'simple'; exponential envelope is applied to the mPDF
118                'full'; actual spin magnitudes are adjusted according to the
119                        correlation length; more accurate (especially for very
120                        short correlation lengths) but slower (especially if
121                        rmax is beyond ~30 A)
122                'auto'; simple method is chosen if xi <= 5 A, full otherwise
123                Note that any other option will be converted to 'simple'
124            linearTermMethod (string): determines how the calculation will
125                handle the linear term present for magnetic structures with a net
126                magnetization. Options are:
127                'exact'; slope will be calculated from the values of
128                    MagStructure.rho0 and MagStructure.netMag, damping set by
129                    MagStructure.corrLength.
130                'autoslope': slope will be determined by least-squares
131                    minimization of the calculated mPDF, thereby ensuring that
132                    the mPDF oscillates around zero, as it is supposed to.
133                    Damping set by MagStructure.corrLength.
134                'fullauto': slope and damping set by least-squares minimization.
135                    This should only be used in the case of an anisotropic
136                    correlation length.
137                Note that any other option will be converted to 'exact'.
138        Returns: numpy array giving the r grid of the calculation, as well as
139            one or both of the mPDF quantities.
140        """
141        peakWidth = np.sqrt(self.magstruc.Uiso)
142        if correlationMethod not in ['simple', 'full', 'auto']:
143            correlationMethod = 'simple'  # convert non-standard inputs to simple
144        if linearTermMethod not in ['exact', 'autoslope', 'fullauto']:
145            linearTermMethod = 'exact'  # convert non-standard inputs to simple
146        dampingMat = self.magstruc.dampingMat
147        xi = self.magstruc.corrLength
148        if correlationMethod == 'auto':  # convert to full or simple
149            if xi <= 5.0:
150                correlationMethod = 'full'
151            else:
152                correlationMethod = 'simple'
153
154        if type(dampingMat) == np.ndarray and correlationMethod != 'full':
155            print("Warning: correlationMethod should be set to 'full' when using")
156            print("the damping matrix instead of a scalar correlation length.")
157        if type(dampingMat) != np.ndarray and linearTermMethod == 'fullauto':
158            print("Warning: 'fullauto' should only be used with an anisotropic")
159            print("correlation length as encoded in the damping matrix.")
160        if (xi==0) and (type(dampingMat) != np.ndarray):
161            calcIdxs = self.magstruc.calcIdxs
162            rcalc, frcalc = calculatemPDF(self.magstruc.atoms, self.magstruc.spins,
163                                          self.magstruc.gfactors, calcIdxs,
164                                          self.rstep, self.rmin, self.rmax,
165                                          peakWidth, self.qmin, self.qmax,
166                                          self.qdamp, self.extendedrmin,
167                                          self.extendedrmax, self.ordScale,
168                                          self.magstruc.K1, self.magstruc.rho0,
169                                          self.magstruc.netMag, xi,
170                                          linearTermMethod, False, self.qwindow,
171                                          self.qgrid)
172        elif correlationMethod == 'full':  # change magnitudes of the spins
173            originalSpins = 1.0*self.magstruc.spins
174            for i, currentIdx in enumerate(self.magstruc.calcIdxs):
175                self.magstruc.spins = self.magstruc.generateScaledSpins(currentIdx)
176                rcalc, frtemp = calculatemPDF(self.magstruc.atoms, self.magstruc.spins,
177                                              self.magstruc.gfactors, [currentIdx],
178                                              self.rstep, self.rmin, self.rmax,
179                                              peakWidth, self.qmin, self.qmax,
180                                              self.qdamp, self.extendedrmin,
181                                              self.extendedrmax, self.ordScale,
182                                              self.magstruc.K1, self.magstruc.rho0,
183                                              self.magstruc.netMag, xi,
184                                              linearTermMethod, False, self.qwindow,
185                                              self.qgrid)
186                if i==0:
187                    frcalc = 1.0*frtemp
188                else:
189                    frcalc += frtemp
190                self.magstruc.spins = 1.0*originalSpins
191            frcalc /= len(self.magstruc.calcIdxs)
192        else:  # simple method: apply exponential envelope
193            calcIdxs = self.magstruc.calcIdxs
194            rcalc, frcalc = calculatemPDF(self.magstruc.atoms, self.magstruc.spins,
195                                          self.magstruc.gfactors, calcIdxs,
196                                          self.rstep, self.rmin, self.rmax,
197                                          peakWidth, self.qmin, self.qmax,
198                                          self.qdamp, self.extendedrmin,
199                                          self.extendedrmax, self.ordScale,
200                                          self.magstruc.K1, self.magstruc.rho0,
201                                          self.magstruc.netMag, xi,
202                                          linearTermMethod, True, self.qwindow,
203                                          self.qgrid)
204        # create a mask to put the calculation on the desired grid
205        mask = np.logical_and(rcalc > self.rmin - 0.5*self.rstep,
206                              rcalc < self.rmax + 0.5*self.rstep)
207        if normalized and not both:
208            return rcalc[mask], frcalc[mask]
209        elif not normalized and not both:
210            Drcalc = calculateDr(rcalc, frcalc, self.magstruc.ffqgrid,
211                                 self.magstruc.ff, self.paraScale, self.rmintr,
212                                 self.rmaxtr, self.rstep, self.qmin, self.qmax,
213                                 self.magstruc.K1, self.magstruc.K2)
214            return rcalc[mask], Drcalc[mask]
215        else:
216            Drcalc = calculateDr(rcalc, frcalc, self.magstruc.ffqgrid,
217                                 self.magstruc.ff, self.paraScale, self.rmintr,
218                                 self.rmaxtr, self.rstep, self.qmin, self.qmax,
219                                 self.magstruc.K1, self.magstruc.K2)
220            return rcalc[mask], frcalc[mask], Drcalc[mask]

222    def plot(self, normalized=True, both=False, scaled=True,
223             correlationMethod='simple', linearTermMethod='exact'):
224        """Plot the magnetic PDF.
225
226        Args:
227            normalized (boolean): indicates whether or not the normalized mPDF
228                should be plotted.
229            both (boolean): indicates whether or not both normalized and
230                unnormalized mPDF quantities should be plotted.
231            correlationMethod (string): determines how the calculation should
232                be done if the correlation length is finite. Options are:
233                'simple'; exponential envelope is applied to the mPDF
234                'full'; actual spin magnitudes are adjusted according to the
235                        correlation length; more accurate (especially for very
236                        short correlation lengths) but slower (especially if
237                        rmax is beyond ~30 A)
238                'auto'; simple method is chosen if xi <= 5 A, full otherwise
239                Note that any other option will be converted to 'simple'
240            linearTermMethod (string): determines how the calculation will
241                handle the linear term present for magnetic structures with a net
242                magnetization. Options are:
243                'exact'; slope will be calculated from the values of
244                    MagStructure.rho0 and MagStructure.netMag, damping set by
245                    MagStructure.corrLength.
246                'autoslope': slope will be determined by least-squares
247                    minimization of the calculated mPDF, thereby ensuring that
248                    the mPDF oscillates around zero, as it is supposed to.
249                    Damping set by MagStructure.corrLength.
250                'fullauto': slope and damping set by least-squares minimization.
251                    This should only be used in the case of an anisotropic
252                    correlation length.
253                Note that any other option will be converted to 'exact'.
254        """
255        fig = plt.figure()
256        ax = fig.add_subplot(111)
257        ax.set_xlabel(r'r ($\mathdefault{\AA}$)')
258        ax.set_xlim(xmin=self.rmin, xmax=self.rmax)
259        if normalized and not both:
260            rcalc, frcalc = self.calc(normalized, both, correlationMethod, linearTermMethod)
261            ax.plot(rcalc, frcalc) 
262            ax.set_ylabel(r'f ($\mathdefault{\AA^{-2}}$)')
263        elif not normalized and not both:
264            rcalc, Drcalc = self.calc(normalized, both, correlationMethod, linearTermMethod)
265            ax.plot(rcalc, Drcalc)
266            ax.set_ylabel(r'd ($\mathdefault{\AA^{-2}}$)')
267        else:
268            rcalc, frcalc, Drcalc = self.calc(normalized, both, correlationMethod, linearTermMethod)
269            if scaled:
270                frscl = np.max(np.abs(frcalc))
271                drscl = np.max(np.abs(Drcalc[rcalc>1.5]))
272                scl = frscl / drscl
273            else:
274                scl = 1.0
275            ax.plot(rcalc, frcalc, 'b-', label='f(r)')
276            ax.plot(rcalc, scl * Drcalc, 'r-', label='d(r)')
277            ax.set_ylabel(r'f, d ($\mathdefault{\AA^{-2}}$)')
278            plt.legend(loc='best')
279        plt.show()

281    def runChecks(self):
282        """Run some quick checks to help with troubleshooting.
283        """
284        print('Running checks on MPDFcalculator...\n')
285
286        flagCount = 0
287        flag = False
288
289        ### check if number of spins and atoms do not match
290        if self.magstruc.atoms.shape[0] != self.magstruc.spins.shape[0]:
291            flag = True
292        if flag:
293            flagCount += 1
294            print('Number of atoms and spins do not match; try calling')
295            print('makeAtoms() and makeSpins() again on your MagStructure.\n')
296        flag = False
297
298        ### check for nan values in spin array
299        if np.any(np.isnan(self.magstruc.spins)):
300            flag = True
301        if flag:
302            flagCount += 1
303            print('Spin array contains nan values ("not a number").\n')
304        flag = False
305
306        ### check if rmax is too big for rmaxAtoms in structure
307        for key in self.magstruc.species:
308            if self.magstruc.species[key].rmaxAtoms < self.rmax:
309                flag = True
310        if flag:
311            flagCount += 1
312            print('Warning: the atoms in your MagStructure may not fill a')
313            print('volume large enough for the desired rmax for the mPDF')
314            print('calculation. Adjust rmax and/or rmaxAtoms in the')
315            print('MagSpecies or MagStructure objects.\n')
316        flag = False
317
318        ### check for unphysical parameters like negative scale factors
319        if np.min((self.paraScale, self.ordScale)) < 0:
320            flag = True
321        if flag:
322            flagCount += 1
323            print('Warning: you have a negative scale factor.')
324            print(('Paramagnetic scale = ', self.paraScale))
325            print(('Ordered scale = ', self.ordScale))
326        flag = False
327
328        if flagCount == 0:
329            print('All checks passed. No obvious problems found.\n')

331    def rgrid(self):
332        """Return the current r grid of the mPDF calculator."""
333        r = np.arange(0,self.rmax+10,self.rstep)        
334        mask = np.logical_and(r > self.rmin - 0.5*self.rstep,
335                              r < self.rmax + 0.5*self.rstep)
336        return r[mask]
