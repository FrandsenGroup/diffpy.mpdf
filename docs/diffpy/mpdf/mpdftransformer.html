 27def resid1(p, ff, iq, diq):
 28    """Scale the magnetic form factor to the data."""
 29    return (iq - p[0] * ff**2)/diq
 30
 31def bkgfunc(p, q):
 32    """Generate polynomial to fit to F(Q)."""
 33    return q * polyval(q, p)
 34
 35def resid2(p, q, fqm, dfqm):
 36    """Fit polynomial to F(Q)."""
 37    return (fqm - bkgfunc(p, q))/dfqm

 39def window_FD(q, qmaxwindow, width):
 40    """
 41    Window function based on the Fermi-Dirac function.
 42    Goes to zero at qmaxwindow.
 43    """
 44    window = 0.0 * q
 45    mask = (q <= qmaxwindow)
 46    window[mask] = 2.0/(np.exp((q[mask]-qmaxwindow)/width) + 1) - 1
 47    return window
 48
 49def window_Lorch(q, qmaxwindow):
 50    """
 51    Lorch window function. Goes to zero at qmaxwindow.
 52    """
 53    window = 0.0 * q
 54    mask = (q <= qmaxwindow)
 55    window[mask] = (qmaxwindow/np.pi/q[mask]) * \
 56                   np.sin(np.pi*q[mask]/qmaxwindow)
 57    return window

 59def getmPDF_unnormalized(q, iq, qmin, qmax, rmin=0.05, rmax=100, rstep=0.01):
 60    """Sine Fourier transform to generate the unnormalized mPDF.
 61
 62    This function is called by an instance of mpdftransformer to generate
 63    the unnormalized mPDF.
 64
 65    Args:
 66        q (np array): Momentum transfer for which scattered intensities
 67            have been measured. Need not be uniform.
 68        iq (np array): Scattered intensity.
 69        qmin (float): Minimum q value included in the Fourier transform.
 70            In inverse Angstroms.
 71        qmax (float): Maximum q value included in the Fourier transform.
 72            In inverse Angstroms.
 73        rmin (float): Minimum r value for the mPDF data. In Angstroms.
 74        rmax (float): Maximum r value for the mPDF data. In Angstroms.
 75        rstep (float): Step size for the r-grid for the mPDF data. In
 76            Angstroms.
 77
 78    Returns:
 79        r (np array): r grid of the unnormalized mPDF.
 80        dmag (np array): Unnormalized mPDF.
 81    """
 82    mask = np.logical_and(q>=qmin, q<=qmax)
 83    q = q[mask]
 84    iq = iq[mask]
 85    r, dmag = sinTransformDirectIntegration(q, q*iq, rmin=rmin, rmax=rmax,
 86                                            rstep=rstep)
 87    dmag *= np.sqrt(2.0/np.pi) # sinTransformDirectIntegration carries factor of sqrt(2/pi), but need 2/pi for mPDF
 88
 89    return r, dmag

 91def getmPDF_normalized(q, iq, ff, qmin, qmax, qmaxinst, rpoly,
 92                  diq=None, qstart=3.0, rmin=0.05, rmax=100, rstep=0.01,
 93                  window='None', qmaxwindow=0, windowedgewidth=0.1,
 94                  ff2scale=0.0):
 95    """PDFgetX3-style processing to produce the normalized mPDF.
 96    
 97    This function is called by an instance of MPDFtransformer to generate
 98    the normalized mPDF and intermediate functions.
 99
100    Args:
101        q (np array): Momentum transfer for which scattered intensities
102            have been measured. Need not be uniform.
103        iq (np array): Scattered intensity.
104        ff (np array): Magnetic form factor on the same grid as q.
105        qmin (float): Minimum q value included in the Fourier transform.
106            In inverse Angstroms.
107        qmax (float): Maximum q value included in the Fourier transform.
108            In inverse Angstroms.
109        qmaxinst (float): Maximum q value included in the background fit.
110            In inverse Angstroms.
111        rpoly (float): real-space distance below which artifacts may be
112            introduced; used with qmaxinst to determin polynomial degree.
113            In Angstroms.
114        diq (np array): Error bars corresponding to the scattered intensity.
115        qstart (float): Starting q-value for which the squared form factor
116            will be fit to the data. Fitting range is [qstart, qmaxinst].
117            In inverse Angstroms.
118        rmin (float): Minimum r value for the mPDF data. In Angstroms.
119        rmax (float): Maximum r value for the mPDF data. In Angstroms.
120        rstep (float): Step size for the r-grid for the mPDF data. In
121            Angstroms.
122        window (str): specifies what type of window function to use. Default
123            is 'None'. Other options are 'FD' for the Fermi-Dirac window
124            and 'Lorch' for the Lorch window.
125        qmaxwindow (float): Sets the q value where the window goes to 0.
126        windowedgewidth (float): approximate width of Fermi-Dirac window.
127            Not applicable to 'Lorch' or 'None' window options.
128        ff2scale (float): scale factor applied to the squared form factor
129            before normalizing. If left as 0.0 by default, the scale factor
130            is determined via least-squares fit to data over the interval
131            from qstart to qmaxinst. Ideally, this scale factor should be
132            equal to (2/3)(gamma_n r_0 / 2)^2 g^2 J(J+1),
133            where (gamma_n r_0 / 2)^2 = 0.07265 in units of barns.
134
135    Returns:
136        Dictionary with the following keys and values:
137        'r' : r-grid of the resulting mPDF data.
138        'gmag': the resulting normalized mPDF data.
139        'sqm': magnetic structure function (no corrections applied)
140        'dsqm': estimated uncertainties for sqm.
141        'fqc': corrected reduced magnetic structure function (this is what
142            gets Fourier transformed).
143        'fqc_prewindow': corrected reduced magnetic structure function
144            before any window function is applied.
145        'fqm': measured (i.e. uncorrected) reduced magnetic structure function.
146        'dfqm': estimated uncertainties for fqm.
147        'windowFunc': the window function used.
148        'ffscale': the scale factor applied to the form factor.
149        'ff2': the scaled, squared magnetic form factor.
150        'bkg': the final polynomial background used.
151        'bkga': the lower-degree polynomial background.
152        'bkgb': the higher-degree polynomial background.
153        'mask1': mask used to scale the squared form factor.
154        'mask2': mask used to fit the polynomial background.
155        'mask3': mask used when computing the Fourier transform.
156    """
157    
158    ### Step 1: Fit squared form factor to asymptotic behavior of I(Q)
159    
160    # set uncertainties to unity if not provided
161    if diq is None:
162        diq = np.ones_like(q)
163    if (diq == np.array([0])).all():
164        diq = np.ones_like(q)
165    
166    # set nans and zeros in diq to large numbers so they have limited weight
167    # in the fits
168    diq[diq==0.0] = 1000*np.nanmax(diq)
169    diq = np.nan_to_num(diq, nan=1000*np.nanmax(diq))
170
171    mask1 = np.logical_and(q>=qstart, q<=qmaxinst)
172    
173    if ff2scale == 0.0:
174        opt1 = least_squares(resid1, [1], args=(ff[mask1], iq[mask1], diq[mask1]))
175
176        ff2scale = opt1.x[0]
177
178    ff = np.sqrt(ff2scale) * ff
179
180    # create a dictionary containing the items to be returned and add the
181    # scaled squared form factor and initial mask to it.
182    rv = {}
183    rv['ff2scale'] = ff2scale
184    rv['ff2'] = ff**2
185    rv['mask1'] = mask1
186
187    ### Step 2: Obtain the measured F(Q), labeled fqm
188    sqm = iq / ff**2
189    dsqm = diq / ff**2
190    fqm = q * (sqm - 1)
191    dfqm = q * dsqm
192    rv['sqm'] = sqm
193    rv['dsqm'] = dsqm
194    rv['fqm'] = fqm
195    rv['dfqm'] = dfqm
196
197
198    ### Step 3: Fit polynomial background to fqm
199    nr = rpoly * qmaxinst / np.pi # non-integer polynomial degree
200    nlow = int(np.floor(nr))
201    nhigh = int(np.ceil(nr))
202    weight_low = nr - nlow
203    weight_high = nhigh - nr
204    
205    mask2 = np.logical_and(q>=qmin, q<=qmaxinst)
206    
207    # fit with lower degree
208    p02a = np.ones(nlow+1)
209    opt2a = least_squares(resid2, p02a, args=(q[mask2], fqm[mask2], dfqm[mask2]))
210
211    bkga = bkgfunc(opt2a.x, q)
212
213    # fit with higher degree
214    p02b = np.ones(nhigh+1)
215    opt2b = least_squares(resid2, p02b, args=(q[mask2], fqm[mask2], dfqm[mask2]))
216
217    bkgb = bkgfunc(opt2b.x, q)
218
219    # now generate the weighted background
220    bkg = weight_low * bkga + weight_high * bkgb
221    
222    # final calculated F(Q)
223    fqc = fqm - bkg
224
225    # add a bunch of other stuff to the return dictionary.
226    rv['mask2'] = mask2
227    rv['bkga'] = bkga
228    rv['bkgb'] = bkgb
229    rv['bkg'] = bkg
230    rv['fqc_prewindow'] = 1.0 * fqc
231
232    if window == 'Lorch':
233        windowFunc = window_Lorch(q, qmaxwindow)
234        fqc *= windowFunc
235    
236    elif window == 'FD':
237        windowFunc = window_FD(q, qmaxwindow, windowedgewidth)
238        fqc *= windowFunc
239
240    else:
241        windowFunc = np.zeros_like(q) + 1.0
242
243    rv['windowFunc'] = windowFunc
244    rv['fqc'] = fqc
245    
246
247    ### Step 4: Compute the Fourier transform
248    mask3 = np.logical_and(q>=qmin, q<=qmax)
249
250    # normalized mPDF
251    r, gmag = sinTransformDirectIntegration(q[mask3], fqc[mask3], rmin=rmin, rmax=rmax, rstep=rstep)
252    gmag *= np.sqrt(2.0/np.pi) # sinTransformDirectIntegration carries factor of sqrt(2/pi), but need 2/pi for mPDF
253
254    rv['r'] = r
255    rv['gmag'] = gmag
256    rv['mask3'] = mask3
257
258    return rv

260class MPDFtransformer:
261    """Control the Fourier transform parameters to produce the mPDF.
262
263
264    This class is used to generate mPDF data (both normalized and unnormalized)
265    from magnetic scattering data. For the normalized mPDF, the magnetic form
266    factor must be provided, and an ad hoc polynomial background correction is
267    applied to minimize slowly varying errors at high q. The algorithm closely
268    follows that used in PDFgetX3 for x-ray PDF data.
269
270    Args:
271        q (np array): Momentum transfer for which scattered intensities
272            have been measured. Need not be uniform.
273        iq (np array): Scattered intensity.
274        ff (np array): Magnetic form factor on the same grid as q.
275        qmin (float): Minimum q value included in the Fourier transform.
276            In inverse Angstroms.
277        qmax (float): Maximum q value included in the Fourier transform.
278            In inverse Angstroms.
279        qmaxinst (float): Maximum q value included in the background fit.
280            In inverse Angstroms.
281        rpoly (float): real-space distance below which artifacts may be
282            introduced; used with qmaxinst to determin polynomial degree.
283            In Angstroms.
284        diq (np array): Error bars corresponding to the scattered intensity.
285        qstart (float): Starting q-value for which the squared form factor
286            will be fit to the data. Fitting range is [qstart, qmaxinst].
287            In inverse Angstroms.
288        rmin (float): Minimum r value for the mPDF data. In Angstroms.
289        rmax (float): Maximum r value for the mPDF data. In Angstroms.
290        rstep (float): Step size for the r-grid for the mPDF data. In
291            Angstroms.
292        window (str): specifies what type of window function to use. Default
293            is 'None'. Other options are 'FD' for the Fermi-Dirac window
294            and 'Lorch' for the Lorch window.
295        qmaxwindow (float): Sets the q value where the window goes to 0.
296        windowedgewidth (float): approximate width of Fermi-Dirac window.
297            Not applicable to 'Lorch' or 'None' window options.
298
299    Properties:
300        r (np array): r grid of the mPDF.
301        dmag (np array): Unnormalized mPDF.
302        gmag (np array): Normalized mPDF.
303        sqm (np array): Magnetic structure function (no corrections applied)
304        dsqm (np array): Estimated uncertainties for sqm.
305        fqc (np array): Corrected reduced magnetic structure function
306            (this is what gets Fourier transformed).
307        fqc_prewindow (np array): Corrected reduced magnetic structure

313        ff2 (np array): The scaled, squared magnetic form factor.
314        bkg (np array): The final polynomial background used.
315        bkga (np array): The lower-degree polynomial background.
316        bkgb (np array): The higher-degree polynomial background.
317        mask1 (np array): Mask used to scale the squared form factor.
318        mask2 (np array): Mask used to fit the polynomial background.
319        mask3 (np array): Mask used when computing the Fourier transform.
320        unnormalized_done (boolean): True if the unnormalized mPDF has
321            been generated.
322        normalized_done (boolean): True if the normalized mPDF has
323            been generated.
324    """

493    def getmPDF(self, type='normalized'):
494        """Generate the magnetic PDF.
495
496        Args:
497            type (str): must be 'normalized' or 'unnormalized'.
498                'normalized': will generate the normalized mPDF.
499                'unnormalized': will generate the unnormalized mPDF.
500        Returns: Doesn't return anything, but populates the relevant
501            MPDFtransformer properties.
502        """
503        if type not in ['normalized', 'unnormalized']:
504            print('Please choose normalized or unnormalized.')
505        elif type == 'unnormalized':
506            r, dmag = getmPDF_unnormalized(self.q, self.iq, self.qmin,
507                                           self.qmax, self.rmin, self.rmax,
508                                           self.rstep)
509            self._r = r
510            self._dmag = dmag
511            self._unnormalized_done = True
512        elif type == 'normalized':
513            output = getmPDF_normalized(self.q, self.iq, self.ff, self.qmin,
514                                        self.qmax, self.qmaxinst, self.rpoly,
515                                        1.0*self.diq, self.qstart, self.rmin,
516                                        self.rmax, self.rstep, self.window,
517                                        self.qmaxwindow, self.windowedgewidth,
518                                        1.0*self.ff2scale)
519            self._r = output['r']
520            self._gmag = output['gmag']
521            self._sqm = output['sqm']
522            self._dsqm = output['dsqm']
523            self._fqc = output['fqc']
524            self._fqc_prewindow = output['fqc_prewindow']
525            self._fqm = output['fqm']
526            self._dfqm = output['dfqm']
527            self._windowFunc = output['windowFunc']
528            self._ff2 = output['ff2']
529            self._bkg = output['bkg']
530            self._bkga = output['bkga']
531            self._bkgb = output['bkgb']
532            self._mask1 = output['mask1']
533            self._mask2 = output['mask2']
534            self._mask3 = output['mask3']
535            self.ff2scale = output['ff2scale']
536            self._normalized_done = True

Control the Fourier transform parameters to produce the mPDF.

This class is used to generate mPDF data (both normalized and unnormalized) from magnetic scattering data. For the normalized mPDF, the magnetic form factor must be provided, and an ad hoc polynomial background correction is applied to minimize slowly varying errors at high q. The algorithm closely follows that used in PDFgetX3 for x-ray PDF data.

Args: q (np array): Momentum transfer for which scattered intensities have been measured. Need not be uniform. iq (np array): Scattered intensity. ff (np array): Magnetic form factor on the same grid as q. qmin (float): Minimum q value included in the Fourier transform. In inverse Angstroms. qmax (float): Maximum q value included in the Fourier transform. In inverse Angstroms. qmaxinst (float): Maximum q value included in the background fit. In inverse Angstroms. rpoly (float): real-space distance below which artifacts may be introduced; used with qmaxinst to determin polynomial degree. In Angstroms. diq (np array): Error bars corresponding to the scattered intensity. qstart (float): Starting q-value for which the squared form factor will be fit to the data. Fitting range is [qstart, qmaxinst]. In inverse Angstroms. rmin (float): Minimum r value for the mPDF data. In Angstroms. rmax (float): Maximum r value for the mPDF data. In Angstroms. rstep (float): Step size for the r-grid for the mPDF data. In Angstroms. window (str): specifies what type of window function to use. Default is 'None'. Other options are 'FD' for the Fermi-Dirac window and 'Lorch' for the Lorch window. qmaxwindow (float): Sets the q value where the window goes to 0. windowedgewidth (float): approximate width of Fermi-Dirac window. Not applicable to 'Lorch' or 'None' window options.
