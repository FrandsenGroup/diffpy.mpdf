 23class Visualizer:
 24    """Store and visualize three-dimensional data.
 25    
 26    Args:
 27        m (numpy array): signal array (3D)
 28        x (numpy array): x-dimension coordinates (1D)
 29        y (numpy array): y-dimension coordinates (1D)
 30        z (numpy array): z-dimension coordinates (1D)
 31    """
 32    def __init__(self, m=None, x=None, y=None, z=None):
 33            
 34        self.m = m
 35        self.x = x
 36        self.y = y
 37        self.z = z
 38        self.dataslice = None # this will become the data slice
 39        if m is None:
 40            self.m = np.array([])
 41        if x is None:
 42            self.x = np.array([])
 43        if y is None:
 44            self.y = np.array([])
 45        if z is None:
 46            self.z = np.array([])
 47        self._sliceAvailable = False
 48        self._a = None # these two instance attributes will construct an arbitrary grid
 49        self._b = None
 50    
 51    @property
 52    def sliceAvailable(self):
 53        """whether or not a data slice has been made"""
 54        return self._sliceAvailable
 55
 56    @property
 57    def a(self):
 58        """internally used grid coordinate"""
 59        return self._a
 60        
 61    @property
 62    def b(self):
 63        """internally used grid coordinate"""
 64        return self._b

 66    def three_points(self, p1, p2, p3):
 67        """find normal vector to the plane created by the three given points
 68        """
 69        
 70        # find two vectors from the three points which lie on the desired plane
 71        vec1 = p2 - p1
 72        vec2 = p3 - p1
 73        # now cross these two vectors to find a vector normal to the plane
 74        normal = np.cross(vec1, vec2)
 75    
 76        # now calculate the centroid of the three points given
 77        x_pos = (p1[0] + p2[0] + p3[0]) / 3
 78        y_pos = (p1[1] + p2[1] + p3[1]) / 3
 79        z_pos = (p1[2] + p2[2] + p3[2]) / 3
 80        cen_pt = np.array([x_pos, y_pos, z_pos])
 81        print('Center Point:', cen_pt)
 82        
 83        return normal, cen_pt

 85    def make_slice(self, len_a=None, len_b=None, dr=None, use_normal=None, 
 86                   cen_pt=None, normal=None, p1=None, p2=None, p3=None,
 87                   returnSlice=False):
 88        """generate a 2D slice through the dataset
 89        
 90        Args:
 91            len_a, len_b (float): the side lengths in Angstroms of the rectangular
 92                slice to be taken through the data
 93            dr (float): determines the spacing of the grid in Angstroms
 94            use_normal (boolean): when True, will create slice from a normal vector and center point. When
 95                False, will create slice from three points
 96            cen_pt (numpy array): the center of the desired slice. Used when use_normal is True
 97            normal (numpy array): the normal vector to desired plane. Used when use_normal is True
 98            p1, p2, p3 (numpy array): three points in 3D space. The desired plane
 99                goes through these points. Used when use_normal is False.
100            returnSlice (boolean): If True, this function will return the slice and grid.
101            
102        Returns:
103            2D array, along with space arrays, representing slice through 3D dataset
104        """
105        if dr is None:
106            dr = 1
107        if use_normal is None:
108            use_normal = True
109        if len_a is None:
110            len_a = 10
111        if len_b is None:
112            len_b = 10
113        if cen_pt is None:
114            cen_pt = np.array([0, 0, 0])
115        if normal is None:
116            normal = np.array([1, 0, 0])
117        if p1 is None:
118            p1 = np.array([0, 1, 0])
119        if p2 is None:
120            p2 = np.array([1, 0, 0])
121        if p3 is None:
122            p3 = np.array([0, 0, 1])
123        
124        # First check if use_normal is False. If so, access three_points function 
125            # to calculate the normal and cen_pt of the desired plane
126        if use_normal is False:
127            normal, cen_pt = self.three_points(p1, p2, p3)
128         
129        # ensure that our basis vector v1 is not the same as normal
130        v1 = np.array([1, 0, 0])
131        if np.allclose(v1, normal):
132            v1 = np.array([0, 1, 0])
133    
134        # now make a matrix which will reflect any vector onto the orthogonal
135            # complement of the normal vec, which is our desired plane
136        # This is done by subtracting from the vector its component along the normal vector
137        m_norm = np.eye(3) - (np.outer(normal, normal.T) / normal.T.dot(normal))
138        
139        # now reflect v1 using m_norm
140        v1 = m_norm.dot(v1)
141        # and create a new vector v2 that is orthogonal to both v1 and normal
142        v2 = np.cross(normal, v1)
143        # we now have 2 vectors to form our plane
144    
145        # now create and normalize Q, which will rotate an arbitrary 
146        # slice to the orientation we desire
147        Q = np.column_stack((v1, v2, np.zeros_like(v1)))
148        Q[:,:2] /= np.linalg.norm(Q[:,:2], axis = 0)
149    
150        # now create an arbitrary slice
151        self._a = np.arange(-len_a / 2, len_a / 2, dr)
152        self._b = np.arange(-len_b / 2, len_b / 2, dr)
153        self._a = np.append(self._a, len_a / 2)
154        self._b = np.append(self._b, len_b / 2)
155        A,B = np.meshgrid(self._a, self._b)
156        locations = np.array([A.reshape(-1), B.reshape(-1), np.zeros(A.size)])  # the slice starts on the x-y plane
157        # now move locations onto our two vectors, and add cen_pt to move slice into position
158        locations = Q.dot(locations).T + (cen_pt)
159    
160        # now we need to interpolate our 3Dmpdf function over this slice
161        points = (self.x, self.y, self.z)
162        interp = interpn(points, self.m, locations)  # list of values of 3Dmpdf at locations
163        self.dataslice = interp.reshape(len(self._b),len(self._a))
164        
165        self._sliceAvailable = True
166        
167        if returnSlice:
168            return self.dataslice, self._a, self._b
169        
170    def visualize(self, cmap='bwr'):
171        """Visualize the current slice.
172        
173        Args:
174            cmap (str): Name of matplotlib colormap to be used for visualization.
175                See https://matplotlib.org/stable/users/explain/colors/colormaps.html
176                for complete list. Default is 'bwr', which works well for
177                3D-PDF data.
178
179        """
180        if self.sliceAvailable:
181            fig = plt.figure()
182            ax = fig.add_subplot(111)
183            ax.set_xlabel(r"$\mathdefault{\AA}$")
184            ax.set_ylabel(r"$\mathdefault{\AA}$")
185            amin, amax = min(self._a), max(self._a)
186            bmin, bmax = min(self._b), max(self._b)
187            im = ax.imshow(self.dataslice,
188                           extent=[amin, amax, bmin, bmax],
189                           cmap=cmap)
190            colorbar(im)
191            #plt.tight_layout()
192            plt.show()
193            
194        else:
195            print('No data slice available for visualization.')
196            print('Please generate a slice using the makeSlice method.')
